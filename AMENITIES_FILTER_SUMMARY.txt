#### **index.html**
```html
<!-- Amenities Filter Section -->
<div class="bg-white p-3 rounded border mb-4">
    <div class="d-flex justify-content-between align-items-center">
        <h6 class="fw-bold mb-0">
            <i class="fas fa-filter text-danger"></i> Lọc theo tiện ích:
        </h6>
        <button type="button" class="btn btn-sm btn-outline-secondary" 
                onclick="clearAmenityFilters()">
            <i class="fas fa-times"></i> Xóa bộ lọc
        </button>
    </div>
    <div id="amenity-filter-container" class="mt-2 d-flex flex-wrap gap-2">
        <!-- Populated by JavaScript -->
    </div>
</div>

<!-- JavaScript -->
<script>
    function toggleAmenityFilter(amenityId) {
        const url = new URL(window.location);
        const amenities = url.searchParams.getAll('amenities');
        
        if (amenities.includes(amenityId.toString())) {
            // Remove amenity
            url.searchParams.delete('amenities');
            amenities.filter(a => a !== amenityId.toString()).forEach(a => {
                url.searchParams.append('amenities', a);
            });
        } else {
            // Add amenity
            url.searchParams.append('amenities', amenityId);
        }
        
        window.location = url.toString();
    }

    function clearAmenityFilters() {
        const url = new URL(window.location);
        url.searchParams.delete('amenities');
        window.location = url.toString();
    }

    function renderAmenityFilters() {
        const container = document.getElementById('amenity-filter-container');
        const amenitiesData = {{ amenities_json|safe }};
        
        amenitiesData.forEach(amenity => {
            const isSelected = selectedAmenities.includes(amenity.id);
            const btnClass = isSelected ? 'btn-danger' : 'btn-outline-danger';
            
            const btn = document.createElement('button');
            btn.type = 'button';
            btn.className = `btn btn-sm ${btnClass}`;
            btn.innerHTML = `<i class="${amenity.icon_class}"></i> ${amenity.name}`;
            btn.onclick = () => toggleAmenityFilter(amenity.id);
            
            container.appendChild(btn);
        });
    }
</script>
```

#### **admin_form.html**
```html
<!-- Amenity Search Box -->
<div class="d-flex justify-content-between align-items-center mb-3">
    <h5 class="fw-bold text-danger mb-0">
        <i class="fas fa-star"></i> Tiện Ích Quán
    </h5>
    <div style="max-width: 250px;">
        <input type="text" id="amenity-search" class="form-control form-control-sm" 
               placeholder="Tìm tiện ích..." onkeyup="filterAmenities()">
    </div>
</div>

<!-- JavaScript Filter -->
<script>
    function filterAmenities() {
        const searchTerm = document.getElementById('amenity-search').value.toLowerCase();
        const amenityCards = document.querySelectorAll('.amenity-card');
        
        amenityCards.forEach(card => {
            const amenityName = card.getAttribute('data-amenity-name');
            if (amenityName.includes(searchTerm)) {
                card.style.display = 'block';
            } else {
                card.style.display = 'none';
            }
        });
    }
</script>
```
