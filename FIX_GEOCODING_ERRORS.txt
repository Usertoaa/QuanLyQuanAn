@csrf_exempt                    # ← THÊM
@require_GET                    # ← THÊM
def api_geocode_address(request):
    query = request.GET.get('q', '').strip()
    
    if not query:
        return JsonResponse({'error': 'Thiếu địa chỉ cần tìm'}, status=400)
    
    try:
        response = requests.get(
            'https://nominatim.openstreetmap.org/search',
            params={
                'format': 'jsonv2',
                'q': query,
                'limit': 5,
                'accept-language': 'vi',
                'countrycodes': 'vn',
                'addressdetails': 1
            },
            headers={
                'User-Agent': 'GIS_Eatery/1.0'
            },
            timeout=10
        )
        response.raise_for_status()
        
        # ↓ CHANGE: Trả về ARRAY tất cả results
        results = response.json()
        if results:
            formatted_results = []
            for result in results:
                formatted_results.append({
                    'lat': float(result.get('lat')),
                    'lon': float(result.get('lon')),  # ← 'lon' not 'lng'
                    'address': result.get('display_name', ''),
                    'display_name': result.get('display_name', '')
                })
            return JsonResponse(formatted_results, safe=False)
        else:
            return JsonResponse([], safe=False)  # ← Return empty array
    
    except requests.RequestException as e:
        print(f'[Geocoding Error] {str(e)}')  # ← Add logging
        return JsonResponse({
            'error': 'Không thể kết nối dịch vụ geocoding...',
            'status': 'error'
        }, status=500)


@csrf_exempt                    # ← ĐÃ CÓ
@require_GET                    # ← ĐÃ CÓ
def api_reverse_geocode_address(request):
    """Chuyển đổi tọa độ (lat, lng) → địa chỉ"""
    lat = request.GET.get('lat', '').strip()
    lng = request.GET.get('lng', '').strip()
    
    if not lat or not lng:
        return JsonResponse({'error': 'Thiếu tọa độ (lat, lng)'}, status=400)
    
    try:
        lat = float(lat)
        lng = float(lng)
    except ValueError:
        return JsonResponse({'error': 'Tọa độ không hợp lệ'}, status=400)
    
    try:
        response = requests.get(
            'https://nominatim.openstreetmap.org/reverse',
            params={
                'format': 'jsonv2',
                'lat': lat,
                'lon': lng,
                'accept-language': 'vi',
                'addressdetails': 1,
                'zoom': 18
            },
            headers={
                'User-Agent': 'GIS_Eatery/1.0'
            },
            timeout=10
        )
        response.raise_for_status()
        return JsonResponse(response.json())
    
    except requests.RequestException as e:
        print(f'[Reverse Geocoding Error] {str(e)}')  # ← Add logging
        return JsonResponse({
            'error': 'Không thể kết nối dịch vụ geocoding...',
            'status': 'error'
        }, status=500)

// ❌ AFTER: Return error object, handle gracefully

async function fetchGeocode(query) {
    const url = `/api/geocode-address/?q=${encodeURIComponent(query)}`;
    try {
        const response = await fetch(url, {
            headers: { 'Accept': 'application/json' }
        });

        if (!response.ok) {
            console.warn(`Geocode API error: ${response.status}`);
            return { error: `HTTP ${response.status}` };
        }

        const data = await response.json();
        // Check if response is array
        if (Array.isArray(data)) {
            return data;
        }
        // Check if error object
        if (data.error) {
            return data;
        }
        // Wrap single object in array
        return Array.isArray(data) ? data : [data];
    } catch (error) {
        console.error('Geocode fetch error:', error);
        return { error: error.message };
    }
}


async function fetchReverseGeocode(lat, lng) {
    const url = `/api/reverse-geocode-address/?lat=${encodeURIComponent(lat)}&lng=${encodeURIComponent(lng)}`;
    try {
        const response = await fetch(url, {
            headers: { 'Accept': 'application/json' }
        });

        if (!response.ok) {
            console.warn(`Reverse geocode API error: ${response.status}`);
            return { error: `HTTP ${response.status}` };
        }

        const data = await response.json();
        return data || { error: 'Không có kết quả' };
    } catch (error) {
        console.error('Reverse geocode fetch error:', error);
        return { error: error.message };
    }
}

async function geocodeAddress() {
    const address = addressInput.value.trim();
    const selectedDistrict = getSelectedDistrictName();

    if (!address) {
        alert('Chưa nhập địa chỉ!');
        return;
    }

    try {
        const queries = [];
        if (selectedDistrict) {
            queries.push(`${address}, ${selectedDistrict}, Thành phố Hồ Chí Minh, Việt Nam`);
        }
        queries.push(`${address}, Thành phố Hồ Chí Minh, Việt Nam`);
        queries.push(address);

        // Try each query until results found
        let results = [];
        for (const query of queries) {
            try {
                const data = await fetchGeocode(query);
                // Check if response is array
                if (Array.isArray(data)) {
                    results = data;
                    if (results.length > 0) break;
                } else if (data && data.error) {
                    console.warn(`Query "${query}" failed: ${data.error}`);
                    continue;
                }
            } catch (err) {
                console.warn(`Query "${query}" error: ${err.message}`);
                continue;
            }
        }

        if (!Array.isArray(results) || results.length === 0) {
            alert('Không tìm thấy địa chỉ. Hãy nhập rõ hoặc nhấp trực tiếp lên bản đồ.');
            return;
        }

        const result = results[0];
        updateMarker(parseFloat(result.lat), parseFloat(result.lon), 18);
        syncDistrictFromResult(result);
    } catch (error) {
        console.error('Geocoding error:', error);
        alert('Lỗi kết nối dịch vụ bản đồ. Nhấp trực tiếp lên bản đồ để chọn vị trí.');
    }
}


// ✅ BEFORE: Not checking for error object
// ❌ AFTER: Check data.error first

async function reverseGeocode(lat, lng) {
    try {
        const data = await fetchReverseGeocode(lat, lng);

        // Check for error
        if (data.error) {
            console.warn('Reverse geocode error:', data.error);
            addressInput.value = `${lat.toFixed(6)}, ${lng.toFixed(6)}`;
            return;
        }

        if (!data) {
            addressInput.value = `${lat.toFixed(6)}, ${lng.toFixed(6)}`;
            return;
        }

        syncDistrictFromResult(data);
        const cleanAddress = buildCleanAddress(data);
        addressInput.value = cleanAddress || `${lat.toFixed(6)}, ${lng.toFixed(6)}`;
    } catch (error) {
        console.error('Reverse geocoding error:', error);
        addressInput.value = `${lat.toFixed(6)}, ${lng.toFixed(6)}`;
    }
}

NOMINATIM API NOTES
================================================================================

✅ Pros:
   - Free & Open Source
   - No API key required
   - Good for Vietnam addresses
   - Multilingual support

⚠️ Cons:
   - Rate limited (1 req/sec)
   - Sometimes slow/timeout
   - Results may vary in quality
   - Can be blocked if too many requests

🔧 Production Alternative:
   - Google Maps API (better accuracy, paid)
   - MapBox (alternative, freemium)
